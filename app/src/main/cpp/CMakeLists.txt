cmake_minimum_required(VERSION 3.22)
project(calltranslate LANGUAGES C CXX)
set(CMAKE_CXX_STANDARD 20)
add_compile_options(-O3)
set(CMAKE_C_FLAGS_DEBUG "-O3 -g" CACHE STRING "" FORCE)
set(CMAKE_CXX_FLAGS_DEBUG "-O3 -g" CACHE STRING "" FORCE)
set(BUILD_SHARED_LIBS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_TESTS OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_EXAMPLES OFF CACHE BOOL "" FORCE)
set(WHISPER_BUILD_SERVER OFF CACHE BOOL "" FORCE)
set(GGML_NATIVE OFF CACHE BOOL "" FORCE)
set(GGML_OPENMP OFF CACHE BOOL "" FORCE)
set(GGML_CPU_ARM_ARCH armv8.2-a+fp16 CACHE STRING "" FORCE)
set(SPM_ENABLE_SHARED OFF CACHE BOOL "" FORCE)
set(SPM_BUILD_TEST OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_TCMALLOC OFF CACHE BOOL "" FORCE)
set(SPM_ENABLE_NFKC_COMPILE OFF CACHE BOOL "" FORCE)
set(CMAKE_POSITION_INDEPENDENT_CODE ON)
if(GGML_VULKAN)
    if(CMAKE_HOST_SYSTEM_NAME STREQUAL "Darwin")
        set(NDK_HOST_TAG darwin-x86_64)
    elseif(CMAKE_HOST_SYSTEM_NAME STREQUAL "Linux")
        set(NDK_HOST_TAG linux-x86_64)
    else()
        set(NDK_HOST_TAG windows-x86_64)
    endif()
    set(Vulkan_INCLUDE_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../../../../vendor/Vulkan-Headers/include" CACHE PATH "" FORCE)
    set(Vulkan_LIBRARY "${CMAKE_ANDROID_NDK}/toolchains/llvm/prebuilt/${NDK_HOST_TAG}/sysroot/usr/lib/aarch64-linux-android/31/libvulkan.so" CACHE FILEPATH "" FORCE)
    set(Vulkan_GLSLC_EXECUTABLE "${CMAKE_ANDROID_NDK}/shader-tools/${NDK_HOST_TAG}/glslc" CACHE FILEPATH "" FORCE)
    set(SPIRV-Headers_DIR "${CMAKE_CURRENT_SOURCE_DIR}/spv-package")
    include_directories("${CMAKE_CURRENT_SOURCE_DIR}/../../../../vendor/SPIRV-Headers/include")
endif()
add_subdirectory(../../../../vendor/whisper.cpp whisper)
add_subdirectory(../../../../vendor/sentencepiece sentencepiece)
add_library(calltranslate SHARED engine.cpp)
target_include_directories(calltranslate PRIVATE ../../../../vendor/sentencepiece/src)
target_link_libraries(calltranslate PRIVATE whisper sentencepiece-static log android)
add_subdirectory(npu)
